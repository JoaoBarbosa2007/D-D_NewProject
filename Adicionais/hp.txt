const statusBars = document.querySelectorAll(".status-bar");

function atualizarStatus(statusBar, valor) {

    const valorMaximo = Number(statusBar.dataset.max);

    valor = Math.max(0, Math.min(valor, valorMaximo));

    statusBar.dataset.current = valor;

    const valorAtual = statusBar.querySelector(".status-value");
    const preenchimento = statusBar.querySelector(".status-fill");

    const porcentagem = (valor / valorMaximo) * 100;

    valorAtual.textContent = `${valor}/${valorMaximo}`;

    preenchimento.style.width = `${porcentagem}%`;
}


function alterarStatus(statusBar, quantidade) {

    const valorAtual = Number(statusBar.dataset.current);

    atualizarStatus(
        statusBar,
        valorAtual + quantidade
    );
}


statusBars.forEach(statusBar => {

    const botaoDiminuirGrande =
        statusBar.querySelector(".decrease-big");

    const botaoDiminuir =
        statusBar.querySelector(".decrease");

    const botaoAumentar =
        statusBar.querySelector(".increase");

    const botaoAumentarGrande =
        statusBar.querySelector(".increase-big");


    botaoDiminuirGrande.addEventListener("click", () => {
        alterarStatus(statusBar, -5);
    });


    botaoDiminuir.addEventListener("click", () => {
        alterarStatus(statusBar, -1);
    });


    botaoAumentar.addEventListener("click", () => {
        alterarStatus(statusBar, 1);
    });


    botaoAumentarGrande.addEventListener("click", () => {
        alterarStatus(statusBar, 5);
    });


    // Inicializa a barra
    atualizarStatus(
        statusBar,
        Number(statusBar.dataset.current)
    );
});
